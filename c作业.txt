12反向输出
#include <stdio.h>

int main() {
    int a;
    scanf("%d",&a);

    if(a>=1000 && a<=9999){
        int qian = a / 1000;        //千位
        int bai  = a / 100 % 10;    //百位
        int shi  = a / 10 % 10;     //十位
        int ge   = a % 10;          //个位

        int fanshu = ge*1000 + shi*100 + bai*10 + qian;
        printf("%04d", fanshu);
    }
    else {
        printf("error");
    }
    return 0;
}
